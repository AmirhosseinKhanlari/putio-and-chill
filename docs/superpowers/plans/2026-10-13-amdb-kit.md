> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `AMDBKit`, the app-state and screen-logic layer that sits between `AMDBCore` and the SwiftUI views. Every piece is unit-tested on macOS with `swift test`, and both the iOS app (Plan 2b) and the tvOS app (Plan 3) reuse it.

**Architecture:** A second library target in the existing `AMDBCore` package. Services are reached through small protocols (`PutioService`, `CatalogService`, `ReleaseSearching`), so tests use fakes. Long-lived app state lives in `@Observable @MainActor` classes:
- `AppModel` is the service graph.
- `LibraryIndex` and `TransferMonitor` track the put.io library and transfers.
- One view model per screen.

Local state persists as JSON through an actor (`LocalStore`). Tokens live in the Keychain.

**Tech Stack:** Swift 6.4, SwiftPM, Observation, Security (Keychain), Swift Testing.

```swift
        }
```
```swift
    }
```

```swift
        lock.withLock {
            nextID += 1

            added.append((url, saveParentID))
```

```swift
import Foundation
import Security

public enum TokenKey: String, CaseIterable, Sendable {
    case putio, tmdb, chill

    /// TMDB and chill tokens sync through iCloud Keychain so the Apple TV app can reuse them.
    var synchronizes: Bool { self != .putio }
}

public protocol TokenStore: Sendable {
    func read(_ key: TokenKey) -> String?
    func write(_ value: String?, for key: TokenKey)
}

public final class InMemoryTokenStore: TokenStore, @unchecked Sendable {
    private let lock = NSLock()
    private var values: [TokenKey: String]

    public init(_ values: [TokenKey: String] = [:]) { self.values = values }
    public func read(_ key: TokenKey) -> String? { lock.withLock { values[key] } }
    public func write(_ value: String?, for key: TokenKey) { lock.withLock { values[key] = value } }
}

public struct KeychainTokenStore: TokenStore {
    private let service: String

    public init(service: String = "app.amdb.tokens") { self.service = service }

    public func read(_ key: TokenKey) -> String? {
        var query = baseQuery(key)
        query[kSecAttrSynchronizable as String] = kSecAttrSynchronizableAny
        query[kSecReturnData as String] = true
        query[kSecMatchLimit as String] = kSecMatchLimitOne
        var result: CFTypeRef?
        guard SecItemCopyMatching(query as CFDictionary, &result) == errSecSuccess,
              let data = result as? Data else { return nil }
        return String(data: data, encoding: .utf8)
    }

    public func write(_ value: String?, for key: TokenKey) {
        var deleteQuery = baseQuery(key)
        deleteQuery[kSecAttrSynchronizable as String] = kSecAttrSynchronizableAny
        SecItemDelete(deleteQuery as CFDictionary)
        guard let value, !value.isEmpty else { return }

        var item = baseQuery(key)
        item[kSecValueData as String] = Data(value.utf8)
        item[kSecAttrAccessible as String] = kSecAttrAccessibleAfterFirstUnlock
        if key.synchronizes { item[kSecAttrSynchronizable as String] = true }
        if SecItemAdd(item as CFDictionary, nil) != errSecSuccess, key.synchronizes {
            // Simulators and free provisioning can refuse synchronizable items; keep a device-only copy.
            item[kSecAttrSynchronizable as String] = false
            SecItemAdd(item as CFDictionary, nil)
        }
    }

    private func baseQuery(_ key: TokenKey) -> [String: Any] {
        [kSecClass as String: kSecClassGenericPassword,
         kSecAttrService as String: service,
         kSecAttrAccount as String: key.rawValue]
    }
}
```

```swift
public actor LocalStore {
    private let fileURL: URL?
    private var state: LocalState

    public init(fileURL: URL?) {
        self.fileURL = fileURL
        if let fileURL, let data = try? Data(contentsOf: fileURL),
           let decoded = try? JSONDecoder().decode(LocalState.self, from: data) {
            state = decoded
        } else {
            state = LocalState()
        }
    }

    public static func defaultURL() -> URL {
        FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appending(path: "AMDB/state.json")
    }

    public var snapshot: LocalState { state }

    public func update(_ body: @Sendable (inout LocalState) -> Void) {
        body(&state)
        persist()
    }

    public func addRecentSearch(_ raw: String) {
        let query = raw.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !query.isEmpty else { return }
        state.recentSearches.removeAll { $0.caseInsensitiveCompare(query) == .orderedSame }
        state.recentSearches.insert(query, at: 0)
        state.recentSearches = Array(state.recentSearches.prefix(10))
        persist()
    }

    private func persist() {
        guard let fileURL else { return }
        try? FileManager.default.createDirectory(at: fileURL.deletingLastPathComponent(), withIntermediateDirectories: true)
        if let data = try? JSONEncoder().encode(state) { try? data.write(to: fileURL, options: .atomic) }
    }
```

```swift
import AMDBCore
import Foundation
import Observation

public struct LibraryEntry: Identifiable, Sendable, Hashable {
    public let file: PutioFile
    public var match: LibraryMatch?
    public let folderName: String?

    public init(file: PutioFile, match: LibraryMatch?, folderName: String?) {
        self.file = file; self.match = match; self.folderName = folderName
    }

    public var id: Int64 { file.id }
    public var startFrom: Int { file.startFrom ?? 0 }
    public var displayTitle: String { match?.title ?? file.name }
}

@Observable @MainActor
public final class LibraryIndex {
    public private(set) var entries: [LibraryEntry] = []
    public private(set) var isRefreshing = false
    public private(set) var lastError: AppError?

    private let store: LocalStore
    private let maxFolders: Int
    private var putio: (any PutioService)?
    private var matcher: LibraryMatcher?
    private var progressOverrides: [Int64: Int] = [:]
    private var touchedAt: [Int64: Date] = [:]

    public init(store: LocalStore, maxFolders: Int = 300) {
        self.store = store
        self.maxFolders = maxFolders
    }

    public func configure(putio: (any PutioService)?, catalog: (any CatalogService)?) {
        self.putio = putio
        matcher = catalog.map { LibraryMatcher(catalog: $0) }
        if putio == nil { entries = [] }
    }

    public func refresh() async {
        guard let putio, !isRefreshing else { return }
        isRefreshing = true
        defer { isRefreshing = false }
        do {
            let found = try await Self.walk(putio, maxFolders: maxFolders)
            let state = await store.snapshot
            var result = found.map { LibraryEntry(file: $0.file, match: state.matches[$0.file.id], folderName: $0.folder) }
            entries = Self.sorted(result)
            lastError = nil
            guard let matcher else { return }
            for index in result.indices where result[index].match == nil && !state.unmatched.contains(result[index].id) {
                let entry = result[index]
                let match = await Self.match(entry, matcher)
                result[index].match = match
                let id = entry.id
                await store.update { state in
                    if let match { state.matches[id] = match } else { state.unmatched.insert(id) }
                }
                if index % 10 == 9 { entries = Self.sorted(result) }
            }
            entries = Self.sorted(result)
        } catch let error as AppError {
            lastError = error
        } catch {
            lastError = .transport(error.localizedDescription)
        }
    }

    public var continueWatching: [LibraryEntry] {
        entries.filter { resumeSeconds(for: $0) >= 30 }.sorted { a, b in
            let ta = touchedAt[a.id], tb = touchedAt[b.id]
            if ta != nil || tb != nil { return (ta ?? .distantPast) > (tb ?? .distantPast) }
            return (a.file.updatedAt ?? "") > (b.file.updatedAt ?? "")
        }
    }

    public var recentlyAdded: [LibraryEntry] { Array(entries.prefix(20)) }

    public func matching(tmdbID: Int, season: Int? = nil, episode: Int? = nil) -> [LibraryEntry] {
        entries.filter {
            $0.match?.tmdbID == tmdbID
                && (season == nil || $0.match?.season == season)
                && (episode == nil || $0.match?.episode == episode)
        }
    }

    public func resumeSeconds(for entry: LibraryEntry) -> Int { progressOverrides[entry.id] ?? entry.startFrom }

    public func noteProgress(fileID: Int64, seconds: Int) {
        progressOverrides[fileID] = seconds
        touchedAt[fileID] = Date()
    }

    public func record(file: PutioFile, match: LibraryMatch) async {
        let id = file.id
        await store.update { state in
            state.matches[id] = match
            state.unmatched.remove(id)
        }
        if let index = entries.firstIndex(where: { $0.id == id }) {
            entries[index].match = match
        } else {
            entries.insert(LibraryEntry(file: file, match: match, folderName: nil), at: 0)
        }
    }

    struct FoundVideo: Sendable { let file: PutioFile; let folder: String? }

    static func walk(_ putio: any PutioService, maxFolders: Int) async throws -> [FoundVideo] {
        var queue: [(id: Int64, name: String?)] = [(0, nil)]
        var visited = 0
        var videos: [FoundVideo] = []
        while !queue.isEmpty, visited < maxFolders {
            let folder = queue.removeFirst()
            visited += 1
            for file in try await putio.listFiles(parentID: folder.id).files {
                if file.isFolder {
                    queue.append((file.id, file.name))
                } else if file.isVideo, !isSample(file) {
                    videos.append(FoundVideo(file: file, folder: folder.name))
                }
            }
        }
        return videos
    }

    static func isSample(_ file: PutioFile) -> Bool {
        file.name.lowercased().contains("sample") && (file.size ?? 0) < 300_000_000
    }

    static func match(_ entry: LibraryEntry, _ matcher: LibraryMatcher) async -> LibraryMatch? {
        if let match = try? await matcher.match(fileName: entry.file.name) { return match }
        guard let folder = entry.folderName, let match = try? await matcher.match(fileName: folder) else { return nil }
        // A folder like "Show S02" names the show; keep any episode number the file name carries.
        let parsed = ReleaseNameParser.parse(entry.file.name)
        return LibraryMatch(tmdbID: match.tmdbID, mediaType: match.mediaType,
                            season: parsed.season ?? match.season, episode: parsed.episode ?? match.episode,
                            confidence: match.confidence * 0.9, title: match.title, year: match.year,
                            posterPath: match.posterPath, backdropPath: match.backdropPath)
    }

    static func sorted(_ entries: [LibraryEntry]) -> [LibraryEntry] {
        entries.sorted { ($0.file.createdAt ?? "") > ($1.file.createdAt ?? "") }
    }
}
```

```swift
import AMDBCore

public enum LibraryFilter: String, CaseIterable, Sendable {
    case all, movies, tv, other

    func includes(_ card: LibraryCard) -> Bool {
        switch (self, card) {
        case (.all, _), (.movies, .movie), (.tv, .show), (.other, .file): true
        default: false
        }
    }
}

public enum LibraryCard: Identifiable, Sendable, Hashable {
    case movie(LibraryEntry)
    case show(tmdbID: Int, title: String, posterPath: String?, episodes: [LibraryEntry])
    case file(LibraryEntry)

    public var id: String {
        switch self {
        case let .movie(entry): "m-\(entry.id)"
        case let .show(tmdbID, _, _, _): "s-\(tmdbID)"
        case let .file(entry): "f-\(entry.id)"
        }
    }

    public static func cards(from entries: [LibraryEntry], filter: LibraryFilter) -> [LibraryCard] {
        var cards: [LibraryCard] = []
        var showIndex: [Int: Int] = [:]
        for entry in entries {
            guard let match = entry.match else {
                cards.append(.file(entry))
                continue
            }
            switch match.mediaType {
            case .movie:
                cards.append(.movie(entry))
            case .tv:
                if let index = showIndex[match.tmdbID], case let .show(id, title, poster, episodes) = cards[index] {
                    cards[index] = .show(tmdbID: id, title: title, posterPath: poster, episodes: episodes + [entry])
                } else {
                    showIndex[match.tmdbID] = cards.count
                    cards.append(.show(tmdbID: match.tmdbID, title: match.title, posterPath: match.posterPath, episodes: [entry]))
                }
            case .person:
                cards.append(.file(entry))
            }
        }
        return cards.filter(filter.includes)
    }
}
```

```swift
import AMDBCore
import Foundation

public enum PlaybackResolver {
    /// The largest video inside `fileID` (the file itself when it is a video), searching up to three folder levels.
    public static func playableFile(_ fileID: Int64, putio: any PutioService) async throws -> PutioFile? {
        let root = try await putio.file(id: fileID)
        if root.isVideo { return root }
        guard root.isFolder else { return nil }
        var best: PutioFile?
        var level = [root.id]
        for _ in 0..<3 where !level.isEmpty {
            var next: [Int64] = []
            for folderID in level {
                for file in try await putio.listFiles(parentID: folderID).files {
                    if file.isFolder {
                        next.append(file.id)
                    } else if file.isVideo, (file.size ?? 0) > (best?.size ?? -1) {
                        best = file
                    }
                }
            }
            level = next
        }
        return best
    }
}

public enum FolderResolver {
    /// Finds or creates `path` (e.g. ["AMDB", "Movies"]) under the put.io root and caches its id.
    public static func folderID(path: [String], putio: any PutioService, store: LocalStore) async throws -> Int64 {
        let key = path.joined(separator: "/")
        if let cached = await store.snapshot.folderIDs[key] { return cached }
        var parent: Int64 = 0
        for name in path {
            let children = try await putio.listFiles(parentID: parent).files
            if let existing = children.first(where: { $0.isFolder && $0.name == name }) {
                parent = existing.id
            } else {
                parent = try await putio.createFolder(name: name, parentID: parent).id
            }
        }
        let id = parent
        await store.update { $0.folderIDs[key] = id }
        return id
    }

    public static func forget(path: [String], store: LocalStore) async {
        let key = path.joined(separator: "/")
        await store.update { $0.folderIDs[key] = nil }
    }
}
```

```swift
import AMDBCore
import Foundation
import Observation

public struct TransferTarget: Sendable, Hashable {
    public let tmdbID: Int
    public let kind: MediaKind
    public let title: String
    public let posterPath: String?
    public let backdropPath: String?
    public let season: Int?

                season: Int?, episode: Int?) {
        self.tmdbID = tmdbID; self.kind = kind; self.title = title; self.posterPath = posterPath
        self.backdropPath = backdropPath; self.season = season; self.episode = episode
    }

    public var folderPath: [String] { kind == .tv ? ["AMDB", "TV", title] : ["AMDB", "Movies"] }
}

public struct TrackedTransfer: Identifiable, Sendable, Equatable {
    public let pending: PendingTransfer
    public var transfer: PutioTransfer?
    public var phase: TransferPhase
    public var isStalled = false
    public var playableFile: PutioFile?
    public var notified = false

    public var id: Int64 { pending.transferID }
    public var percent: Int { transfer?.percentDone ?? 0 }
}

public enum TransferEvent: Sendable, Equatable {
    case ready(TrackedTransfer)
    case failed(TrackedTransfer)
}

@Observable @MainActor
public final class TransferMonitor {
    public private(set) var tracked: [TrackedTransfer] = []
    @ObservationIgnored public var onEvent: (@MainActor (TransferEvent) -> Void)?

    private let store: LocalStore
    private let pollInterval: Duration
    private let now: @Sendable () -> Date
    private var putio: (any PutioService)?
    @ObservationIgnored private weak var library: LibraryIndex?
    private var detectors: [Int64: StallDetector] = [:]
    private var loop: Task<Void, Never>?

    public init(store: LocalStore, pollInterval: Duration = .seconds(2), now: @escaping @Sendable () -> Date = { Date() }) {
        self.store = store
        self.pollInterval = pollInterval
        self.now = now
    }

    public func configure(putio: (any PutioService)?, library: LibraryIndex?) {
        self.putio = putio
        self.library = library
        if putio == nil {
            stop()
            tracked = []
        }
    }

    /// Picks up transfers that were in flight when the app last quit.
    public func restore() async {
        for pending in await store.snapshot.pending where !tracked.contains(where: { $0.id == pending.transferID }) {
            tracked.append(TrackedTransfer(pending: pending, transfer: nil, phase: .queued))
        }
        await pollOnce()
        startIfNeeded()
    }

    @discardableResult
    public func send(_ release: Release, target: TransferTarget) async throws -> TrackedTransfer {
        guard let putio else { throw AppError.unauthorized(service: "putio") }
        let transfer: PutioTransfer
        do {
            let folder = try await FolderResolver.folderID(path: target.folderPath, putio: putio, store: store)
            transfer = try await putio.addTransfer(url: release.link, saveParentID: folder)
        } catch AppError.http {
            // The cached folder may have been deleted in put.io; resolve it again once.
            await FolderResolver.forget(path: target.folderPath, store: store)
            let folder = try await FolderResolver.folderID(path: target.folderPath, putio: putio, store: store)
            transfer = try await putio.addTransfer(url: release.link, saveParentID: folder)
        }
        let pending = PendingTransfer(transferID: transfer.id, tmdbID: target.tmdbID, kind: target.kind, title: target.title,
                                      posterPath: target.posterPath, backdropPath: target.backdropPath,
                                      season: target.season, episode: target.episode, createdAt: now())
        await store.update { $0.pending.append(pending) }
        tracked.append(TrackedTransfer(pending: pending, transfer: transfer, phase: TransferPhase(transfer)))
        await settle(transfer.id)
        startIfNeeded()
        return tracked.first { $0.id == transfer.id } ?? TrackedTransfer(pending: pending, transfer: transfer, phase: TransferPhase(transfer))
    }

    public func pollOnce() async {
        guard let putio else { return }
        for id in tracked.filter({ !$0.notified }).map(\.id) {
            guard let transfer = try? await putio.transfer(id: id),
                  let index = tracked.firstIndex(where: { $0.id == id }) else { continue }
            var detector = detectors[id] ?? StallDetector()
            tracked[index].transfer = transfer
            tracked[index].phase = TransferPhase(transfer)
            tracked[index].isStalled = detector.isStalled(transfer, now: now())
            detectors[id] = detector
            await settle(id)
        }
    }

    public func cancel(_ id: Int64) async throws {
        try await putio?.cancelTransfers(ids: [id])
        tracked.removeAll { $0.id == id }
        await store.update { $0.pending.removeAll { $0.transferID == id } }
    }

    public func dismiss(_ id: Int64) { tracked.removeAll { $0.id == id } }

    public func latest(tmdbID: Int, season: Int? = nil, episode: Int? = nil) -> TrackedTransfer? {
        tracked.last { $0.pending.tmdbID == tmdbID && $0.pending.season == season && $0.pending.episode == episode }
    }

    public func stop() {
        loop?.cancel()
        loop = nil
    }

    private func startIfNeeded() {
        guard loop == nil, tracked.contains(where: { !$0.notified }) else { return }
        loop = Task { [weak self] in
            while let self, !Task.isCancelled, self.tracked.contains(where: { !$0.notified }) {
                try? await Task.sleep(for: self.pollInterval)
                await self.pollOnce()
            }
            self?.loop = nil
        }
    }

    /// Resolves a finished transfer to its playable file, records it in the library, and emits one event.
    private func settle(_ id: Int64) async {
        guard let putio, let item = tracked.first(where: { $0.id == id }), !item.notified else { return }
        switch item.phase {
        case let .completed(fileID?):
            guard let file = try? await PlaybackResolver.playableFile(fileID, putio: putio),
                  let index = tracked.firstIndex(where: { $0.id == id }) else { return }
            tracked[index].playableFile = file
            tracked[index].notified = true
            let pending = item.pending
            await library?.record(file: file, match: LibraryMatch(
                tmdbID: pending.tmdbID, mediaType: pending.kind, season: pending.season, episode: pending.episode,
                confidence: 1, title: pending.title, posterPath: pending.posterPath, backdropPath: pending.backdropPath))
            await store.update { $0.pending.removeAll { $0.transferID == id } }
            if let done = tracked.first(where: { $0.id == id }) { onEvent?(.ready(done)) }
        case .failed:
            guard let index = tracked.firstIndex(where: { $0.id == id }) else { return }
            tracked[index].notified = true
            await store.update { $0.pending.removeAll { $0.transferID == id } }
            if let failed = tracked.first(where: { $0.id == id }) { onEvent?(.failed(failed)) }
        default:
            break
        }
    }
}
```

```swift
import AMDBCore
import Foundation
import Observation

public struct ServiceFactory: Sendable {
    public var makePutio: @Sendable (String) -> any PutioService
    public var makeCatalog: @Sendable (String) -> any CatalogService
    public var makeProvider: @Sendable (SourceConfig, _ chillToken: String?) -> (any SourceProvider)?

    public init(makePutio: @escaping @Sendable (String) -> any PutioService,
                makeCatalog: @escaping @Sendable (String) -> any CatalogService,
                makeProvider: @escaping @Sendable (SourceConfig, String?) -> (any SourceProvider)?) {
        self.makePutio = makePutio
        self.makeCatalog = makeCatalog
        self.makeProvider = makeProvider
    }

    public static let live: ServiceFactory = {
        let transport = URLSessionTransport()
        return ServiceFactory(
            makePutio: { PutioClient(token: $0, transport: transport) },
            makeCatalog: { CatalogClient(accessToken: $0, transport: transport) },
            makeProvider: { config, chillToken in
                guard let url = URL(string: config.baseURL) else { return nil }
                switch config.id {
                case "chill": return chillToken.map { ChillProvider(token: $0, transport: transport, baseURL: url) }
                case "yts": return YTSProvider(baseURL: url, transport: transport)
                case "eztv": return EZTVProvider(baseURL: url, transport: transport)
                case "apibay": return ApibayProvider(baseURL: url, transport: transport)
                case "knaben": return KnabenProvider(baseURL: url, transport: transport)
                default: return nil
                }
            })
    }()
}

@Observable @MainActor
public final class AppModel {
    public private(set) var putio: (any PutioService)?
    public private(set) var catalog: (any CatalogService)?
    public private(set) var settings: AppSettings
    public private(set) var hasChill = false
    public let library: LibraryIndex
    public let transfers: TransferMonitor
    public let store: LocalStore

    @ObservationIgnored private let tokens: any TokenStore
    @ObservationIgnored private let userDefaults: UserDefaults
    @ObservationIgnored private let factory: ServiceFactory
    @ObservationIgnored private var searcher: any ReleaseSearching = ReleaseAggregator(providers: [])

    public init(tokens: any TokenStore, defaults: UserDefaults = .standard, store: LocalStore, factory: ServiceFactory = .live) {
        self.tokens = tokens
        self.userDefaults = defaults
        self.store = store
        self.factory = factory
        settings = AppSettings.load(from: defaults)
        library = LibraryIndex(store: store)
        transfers = TransferMonitor(store: store)
        rebuild()
    }

    public var isReady: Bool { putio != nil && catalog != nil }
    public var releaseSearcher: any ReleaseSearching { searcher }

    public func token(_ key: TokenKey) -> String? { tokens.read(key) }

    public func setToken(_ value: String?, for key: TokenKey) {
        tokens.write(value, for: key)
        rebuild()
    }

    public func updateSettings(_ newSettings: AppSettings) {
        settings = newSettings
        newSettings.save(to: userDefaults)
        rebuild()
    }

    public func validateTMDB(_ token: String) async throws {
        try await factory.makeCatalog(token).validate()
    }

    public func notePlayed(_ title: PlayedTitle) async {
        await store.update { $0.lastPlayed = title }
    }

    /// Lowercased infohashes of finished put.io transfers, used to pin "already in put.io" versions.
    public func completedHashes() async -> Set<String> {
        guard let putio, let transfers = try? await putio.transfers() else { return [] }
        return Set(transfers.compactMap { transfer in
            guard case .completed = TransferPhase(transfer) else { return nil }
            return transfer.hash?.lowercased()
        })
    }

    private func rebuild() {
        putio = tokens.read(.putio).map(factory.makePutio)
        catalog = tokens.read(.tmdb).map(factory.makeCatalog)
        let chillToken = tokens.read(.chill)
        hasChill = chillToken != nil
        let providers = settings.sources.filter(\.enabled).compactMap { factory.makeProvider($0, chillToken) }
        searcher = ReleaseAggregator(providers: providers)
        library.configure(putio: putio, catalog: catalog)
        transfers.configure(putio: putio, library: library)
    }
}
```

```swift
import AMDBCore
import Foundation

/// Bridges the player to put.io's `start_from`: periodic saves, and a reset once the video is basically over.
@MainActor
public final class PlaybackSession {
    public let fileID: Int64
    public let resumeFrom: Int
    private let putio: any PutioService
    private weak var library: LibraryIndex?
    private let saveEvery: Int
    private var lastSaved: Int

    public init(fileID: Int64, resumeFrom: Int, putio: any PutioService, library: LibraryIndex?, saveEvery: Int = 10) {
        self.fileID = fileID
        self.resumeFrom = resumeFrom
        self.putio = putio
        self.library = library
        self.saveEvery = saveEvery
        self.lastSaved = resumeFrom
    }

    public var shouldOfferResume: Bool { resumeFrom > 30 }
    public var streamURL: URL { putio.hlsURL(fileID: fileID) }

    public func tick(seconds: Double, duration: Double) async {
        let position = Int(seconds)
        guard abs(position - lastSaved) >= saveEvery else { return }
        await save(position, duration: duration)
    }

    public func finish(seconds: Double, duration: Double) async {
        await save(Int(seconds), duration: duration)
    }

    private func save(_ position: Int, duration: Double) async {
        lastSaved = position
        let watchedToEnd = duration > 0 && Double(position) >= duration * 0.95
        let value = watchedToEnd ? 0 : position
        library?.noteProgress(fileID: fileID, seconds: value)
        try? await putio.setStartFrom(fileID: fileID, seconds: value)
    }
}
```

```swift
import Foundation

public struct CatalogRow: Identifiable, Sendable, Equatable {
    public let id: String
    public let title: String
    public let items: [CatalogItem]
    public init(id: String, title: String, items: [CatalogItem]) { self.id = id; self.title = title; self.items = items }
}

public struct Genre: Sendable, Hashable {
    public let id: Int
    public let name: String
    public init(id: Int, name: String) { self.id = id; self.name = name }
}

@Observable @MainActor
public final class HomeModel {
    public private(set) var hero: [CatalogItem] = []
    public private(set) var rows: [CatalogRow] = []
    public private(set) var isLoading = false
    public private(set) var error: AppError?

    private let catalog: any CatalogService
    private let store: LocalStore

    public init(catalog: any CatalogService, store: LocalStore) {
        self.catalog = catalog
        self.store = store
    }

    public func load() async {
        isLoading = true
        defer { isLoading = false }
        let catalog = catalog
        async let trending = try? catalog.trending(nil)
        async let movies = try? catalog.popular(.movie)
        async let shows = try? catalog.popular(.tv)
        let lastPlayed = await store.snapshot.lastPlayed
        var recommendations: [CatalogItem]?
        if let lastPlayed {
            recommendations = try? await catalog.detail(lastPlayed.kind, id: lastPlayed.tmdbID).recommendations
        }
        let (trendingItems, movieItems, showItems) = await (trending, movies, shows)

        let visibleTrending = (trendingItems ?? []).filter { $0.mediaType != .person }
        hero = Array(visibleTrending.filter { $0.backdropPath != nil }.prefix(6))
        var built: [CatalogRow] = []
        if !visibleTrending.isEmpty { built.append(CatalogRow(id: "trending", title: "Trending Now", items: visibleTrending)) }
        if let lastPlayed, let recommendations, !recommendations.isEmpty {
            built.append(CatalogRow(id: "because", title: "Because you watched \(lastPlayed.title)", items: recommendations))
        }
        if let movieItems, !movieItems.isEmpty { built.append(CatalogRow(id: "movies", title: "Popular Movies", items: movieItems)) }
        if let showItems, !showItems.isEmpty { built.append(CatalogRow(id: "tv", title: "Popular TV", items: showItems)) }
        rows = built
        error = (trendingItems == nil && movieItems == nil && showItems == nil) ? .transport("Couldn't reach TMDB") : nil
    }
}

public final class BrowseModel {
    public let kind: MediaKind
    public private(set) var rows: [CatalogRow] = []
    public private(set) var isLoading = false
    private let catalog: any CatalogService

    public init(kind: MediaKind, catalog: any CatalogService) {
        self.kind = kind
        self.catalog = catalog
    }

    public static func genres(for kind: MediaKind) -> [Genre] {
        switch kind {
        case .tv:
            [Genre(id: 18, name: "Drama"), Genre(id: 35, name: "Comedy"), Genre(id: 80, name: "Crime"),
             Genre(id: 10765, name: "Sci-Fi & Fantasy"), Genre(id: 9648, name: "Mystery"),
             Genre(id: 16, name: "Animation"), Genre(id: 99, name: "Documentary"), Genre(id: 10764, name: "Reality")]
        default:
            [Genre(id: 28, name: "Action"), Genre(id: 35, name: "Comedy"), Genre(id: 18, name: "Drama"),
             Genre(id: 878, name: "Science Fiction"), Genre(id: 53, name: "Thriller"), Genre(id: 27, name: "Horror"),
             Genre(id: 16, name: "Animation"), Genre(id: 99, name: "Documentary")]
        }
    }

    public func load() async {
        isLoading = true
        defer { isLoading = false }
        let catalog = catalog
        let kind = kind
        let genres = Self.genres(for: kind)
        let loaded = await withTaskGroup(of: (Int, CatalogRow?).self) { group in
            group.addTask { (0, (try? await catalog.popular(kind)).map { CatalogRow(id: "popular", title: "Popular", items: $0) }) }
            group.addTask { (1, (try? await catalog.topRated(kind)).map { CatalogRow(id: "top", title: "Top Rated", items: $0) }) }
            for (offset, genre) in genres.enumerated() {
                group.addTask {
                    (offset + 2, (try? await catalog.discover(kind, genreID: genre.id))
                        .map { CatalogRow(id: "g\(genre.id)", title: genre.name, items: $0) })
                }
            }
            var byOrder: [Int: CatalogRow] = [:]
            for await (order, row) in group { if let row, !row.items.isEmpty { byOrder[order] = row } }
            return byOrder.keys.sorted().compactMap { byOrder[$0] }
        }
        rows = loaded
    }
}
```

```swift
import AMDBCore
import Foundation
import Observation

@Observable @MainActor
public final class SearchModel {
    public var query = "" {
        didSet { if query != oldValue { schedule() } }
    }
    public private(set) var results: [CatalogItem] = []
    public private(set) var isSearching = false
    public private(set) var recent: [String] = []

    @ObservationIgnored private let engine: LiveSearchEngine<CatalogItem>
    @ObservationIgnored private let store: LocalStore
    @ObservationIgnored private var task: Task<Void, Never>?

    public init(catalog: any CatalogService, store: LocalStore, debounce: Duration = .milliseconds(250)) {
        self.store = store
        engine = LiveSearchEngine(debounce: debounce) { query in try await catalog.searchMulti(query) }
    }

    private var trimmed: String { query.trimmingCharacters(in: .whitespacesAndNewlines) }

    public var suggestions: [CatalogItem] { Array(results.filter { $0.mediaType != .person }.prefix(6)) }

    public var grid: [CatalogItem] {
        var seen = Set<String>()
        var items: [CatalogItem] = []
        for item in results {
            let expanded = item.mediaType == .person ? item.knownFor.filter { $0.mediaType != .person } : [item]
            for candidate in expanded where seen.insert(candidate.id).inserted { items.append(candidate) }
        }
        return items
    }

    public var releaseQuery: MediaQuery? { trimmed.isEmpty ? nil : .text(trimmed) }

    public func loadRecent() async { recent = await store.snapshot.recentSearches }

    public func commit() async {
        await store.addRecentSearch(trimmed)
        await loadRecent()
    }

    private func schedule() {
        task?.cancel()
        let text = trimmed
        guard !text.isEmpty else {
            results = []
            isSearching = false
            return
        }
        isSearching = true
        let engine = engine
        task = Task {
            guard let found = try? await engine.results(for: text), !Task.isCancelled else { return }
            results = found
            isSearching = false
        }
    }
}
```

```swift
import AMDBCore
import Foundation
import Observation

public enum TitleAction: Equatable, Sendable {
    case get
    case play(LibraryEntry)
    case resume(LibraryEntry, seconds: Int)
    case downloading(TrackedTransfer)
    case failed(TrackedTransfer)
}

@Observable @MainActor
public final class TitleModel {
    public let item: CatalogItem
    public private(set) var detail: TitleDetail?
    public private(set) var episodes: [Episode] = []
    public private(set) var selectedSeason: Int?
    public private(set) var error: AppError?

    private let catalog: any CatalogService
    private let library: LibraryIndex
    private let transfers: TransferMonitor

    public init(item: CatalogItem, catalog: any CatalogService, library: LibraryIndex, transfers: TransferMonitor) {
        self.item = item
        self.catalog = catalog
        self.library = library
        self.transfers = transfers
    }

    public func load() async {
        do {
            let loaded = try await catalog.detail(item.mediaType, id: item.tmdbID)
            detail = loaded
            error = nil
            if item.mediaType == .tv, let first = loaded.seasons.first?.number {
                await selectSeason(first)
            }
        } catch let failure as AppError {
            error = failure
        } catch {
            self.error = .transport(error.localizedDescription)
        }
    }

    public func selectSeason(_ number: Int) async {
        selectedSeason = number
        episodes = (try? await catalog.season(tvID: item.tmdbID, number: number)) ?? []
    }

    public func action(season: Int? = nil, episode: Int? = nil) -> TitleAction {
        if let transfer = transfers.latest(tmdbID: item.tmdbID, season: season, episode: episode) {
            switch transfer.phase {
            case .failed: return .failed(transfer)
            case .completed where transfer.playableFile != nil: break
            default: return .downloading(transfer)
            }
        }
        let hits = library.matching(tmdbID: item.tmdbID, season: season, episode: episode)
        guard let best = hits.max(by: { library.resumeSeconds(for: $0) < library.resumeSeconds(for: $1) }) else { return .get }
        let seconds = library.resumeSeconds(for: best)
        return seconds >= 30 ? .resume(best, seconds: seconds) : .play(best)
    }

    public func query(season: Int? = nil, episode: Int? = nil) -> MediaQuery? {
        guard let detail, let imdbID = detail.imdbID else { return nil }
        let title = detail.item.title
        switch (item.mediaType, season, episode) {
        case (.tv, let season?, let episode?): return .episode(imdbID: imdbID, title: title, season: season, episode: episode)
        case (.tv, let season?, nil): return .season(imdbID: imdbID, title: title, season: season)
        case (.movie, _, _): return .movie(imdbID: imdbID, title: title, year: detail.item.year)
        default: return nil
        }
    }

    public func target(season: Int? = nil, episode: Int? = nil) -> TransferTarget {
        TransferTarget(tmdbID: item.tmdbID, kind: item.mediaType, title: detail?.item.title ?? item.title,
                       posterPath: item.posterPath, backdropPath: item.backdropPath, season: season, episode: episode)
    }
}
```

```swift
import AMDBCore
import Foundation
import Observation

@Observable @MainActor
public final class VersionsModel {
    public let query: MediaQuery
    public let playNow: [LibraryEntry]
    public private(set) var snapshot = ReleaseSnapshot()
    public private(set) var isLoading = false
    public var resolution: Resolution?
    public var sort: ReleaseSort = .seeders

    @ObservationIgnored private let searcher: any ReleaseSearching
    @ObservationIgnored private let completedHashes: Set<String>
    @ObservationIgnored private var task: Task<Void, Never>?

    public init(query: MediaQuery, searcher: any ReleaseSearching, playNow: [LibraryEntry],
                completedHashes: Set<String>, defaultResolution: Resolution?) {
        self.query = query
        self.searcher = searcher
        self.playNow = playNow
        self.completedHashes = completedHashes
        self.resolution = defaultResolution
    }

    public var rows: [Release] { snapshot.releases.filtered(resolution: resolution).sorted(by: sort) }
    public var failures: [ProviderFailure] { snapshot.failures }

    public var availableResolutions: [Resolution] {
        Set(snapshot.releases.map(\.parsed.resolution)).filter { $0 != .unknown }.sorted(by: >)
    }

    public func isInPutio(_ release: Release) -> Bool {
        release.infoHash.map(completedHashes.contains) ?? false
    }

    public func start() {
        task?.cancel()
        snapshot = ReleaseSnapshot()
        isLoading = true
        let stream = searcher.stream(for: query)
        task = Task {
            for await next in stream {
                snapshot = next
                isLoading = !next.isComplete
            }
            isLoading = false
        }
    }

    public func cancel() {
        task?.cancel()
        task = nil
        isLoading = false
    }
}
```
