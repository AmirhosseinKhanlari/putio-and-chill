# AMDBCore Implementation Plan (Plan 1 of 2)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `AMDBCore`, a Swift package containing every piece of non-UI logic the AMDB iPad app needs. Each piece is unit-tested. The package also includes a live verification CLI (`amdb-spike`) that checks the spec's open questions against the real services.

**Architecture:** A SwiftPM library. All networking goes through a small `HTTPTransport` protocol, so tests inject a stub and never touch the network. Torrent sources implement one `SourceProvider` protocol. `ReleaseAggregator` fans out to the sources, merges their results, and streams them. `PutioClient` and `CatalogClient` are thin typed REST wrappers. Pure logic (parser, magnet helpers, matcher, stall detector, search debouncer) has no I/O. Plan 2 (the SwiftUI app) consumes this package.

**Tech Stack:** Swift 6.4 (Swift 6 language mode), SwiftPM, Foundation, Swift Testing (`import Testing`), AVFoundation (spike only). There are no third-party dependencies.

```swift
    convenience init(status: Int = 200, json: String) {
        let body = Data(json.utf8)
        self.init { _ in (status, body) }
    }

        lock.withLock { recorded.append(request) }
        let (status, data) = try handler(request)
        let response = HTTPURLResponse(url: request.url!, statusCode: status, httpVersion: nil, headerFields: nil)!
        return (data, response)
    }
}
```

```swift
import Foundation

public protocol HTTPTransport: Sendable {
    func send(_ request: URLRequest) async throws -> (Data, HTTPURLResponse)
}

public struct URLSessionTransport: HTTPTransport {
    private let session: URLSession
    public init(session: URLSession = .shared) { self.session = session }

    public func send(_ request: URLRequest) async throws -> (Data, HTTPURLResponse) {
        let (data, response) = try await session.data(for: request)
        guard let http = response as? HTTPURLResponse else { throw AppError.transport("non-HTTP response") }
        return (data, http)
    }
}

public enum AppError: Error, Equatable, Sendable {
    case unauthorized(service: String)
    case http(service: String, status: Int)
    case decoding(service: String, detail: String)
    case transport(String)
    case timeout(service: String)
}

public struct HTTPRequester: Sendable {
    public let service: String
    public let transport: any HTTPTransport

    public init(service: String, transport: any HTTPTransport) {
        self.service = service
        self.transport = transport
    }

    public func data(_ request: URLRequest) async throws -> Data {
        let result: (Data, HTTPURLResponse)
        do {
            result = try await transport.send(request)
        } catch let error as AppError {
            throw error
        } catch is CancellationError {
            throw CancellationError()
        } catch let error as URLError where error.code == .cancelled {
            throw CancellationError()
        } catch let error as URLError where error.code == .timedOut {
            throw AppError.timeout(service: service)
        } catch {
            throw AppError.transport("\(service): \(error.localizedDescription)")
        }
        let (data, response) = result
        switch response.statusCode {
        case 200..<300: return data
        case 401: throw AppError.unauthorized(service: service)
        default: throw AppError.http(service: service, status: response.statusCode)
        }
    }

    public func decode<T: Decodable>(_ type: T.Type, _ request: URLRequest, decoder: JSONDecoder = JSONDecoder()) async throws -> T {
        let body = try await data(request)
        do {
            return try decoder.decode(T.self, from: body)
        } catch {
            throw AppError.decoding(service: service, detail: String(describing: error))
        }
    }
}

extension JSONDecoder {
    public static var snakeCase: JSONDecoder {
        let decoder = JSONDecoder()
        decoder.keyDecodingStrategy = .convertFromSnakeCase
        return decoder
    }
}

extension URLRequest {
    public static func get(_ url: URL, headers: [String: String] = [:]) -> URLRequest {
        var request = URLRequest(url: url)
        request.httpMethod = "GET"
        request.setValue("application/json", forHTTPHeaderField: "Accept")
        for (key, value) in headers { request.setValue(value, forHTTPHeaderField: key) }
        return request
    }

    public static func postJSON(_ url: URL, body: some Encodable, headers: [String: String] = [:]) throws -> URLRequest {
        var request = get(url, headers: headers)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpBody = try JSONEncoder().encode(body)
        return request
    }

    public static func postForm(_ url: URL, fields: [(String, String)], headers: [String: String] = [:]) -> URLRequest {
        var request = get(url, headers: headers)
        request.httpMethod = "POST"
        request.setValue("application/x-www-form-urlencoded", forHTTPHeaderField: "Content-Type")
        request.httpBody = Data(fields.map { "\(formEncode($0.0))=\(formEncode($0.1))" }.joined(separator: "&").utf8)
        return request
    }
}

let unreservedCharacters = CharacterSet(charactersIn: "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-._~")

func formEncode(_ value: String) -> String {
    value.addingPercentEncoding(withAllowedCharacters: unreservedCharacters) ?? value
}

/// Decodes an integer sent either as a JSON number or a numeric string (proto3 int64, EZTV, apibay).
public struct LossyInt64: Decodable, Sendable, Hashable {
    public let value: Int64?
    public init(_ value: Int64?) { self.value = value }
    public init(from decoder: any Decoder) throws {
        let container = try decoder.singleValueContainer()
        if let number = try? container.decode(Int64.self) {
            value = number
        } else if let text = try? container.decode(String.self) {
            value = Int64(text)
        } else {
            value = nil
        }
    }
}
```

```swift
import Foundation

public enum MediaQuery: Sendable, Hashable {
    case movie(imdbID: String, title: String, year: Int?)
    case episode(imdbID: String, title: String, season: Int, episode: Int)
    case season(imdbID: String, title: String, season: Int)
    case text(String)

    public var searchText: String {
        switch self {
        case let .movie(_, title, year): year.map { "\(title) \($0)" } ?? title
        case let .episode(_, title, season, episode): "\(title) S\(twoDigits(season))E\(twoDigits(episode))"
        case let .season(_, title, season): "\(title) S\(twoDigits(season))"
        case let .text(text): text
        }
    }

    public var imdbID: String? {
        switch self {
        case let .movie(id, _, _), let .episode(id, _, _, _), let .season(id, _, _): id
        case .text: nil
        }
    }
}

func twoDigits(_ n: Int) -> String { n < 10 ? "0\(n)" : "\(n)" }
```

```swift
import Foundation

public enum Magnet {
    public static let defaultTrackers = [
        "udp://tracker.opentrackr.org:1337/announce",
        "udp://open.stealth.si:80/announce",
        "udp://tracker.torrent.eu.org:451/announce",
        "udp://exodus.desync.com:6969/announce",
        "udp://tracker.openbittorrent.com:6969/announce",
    ]

    public static func infoHash(from link: String) -> String? {
        guard link.lowercased().hasPrefix("magnet:"),
              let match = link.firstMatch(of: /(?i)xt=urn:btih:([a-z0-9]+)/) else { return nil }
        return normalize(String(match.1))
    }

    /// Returns a lowercased 40-char hex infohash from hex or base32 input.
    public static func normalize(_ hash: String) -> String? {
        let trimmed = hash.trimmingCharacters(in: .whitespaces)
        if trimmed.count == 40, trimmed.allSatisfy(\.isHexDigit) { return trimmed.lowercased() }
        if trimmed.count == 32 { return base32ToHex(trimmed.uppercased()) }
        return nil
    }

    public static func make(infoHash: String, name: String, trackers: [String] = defaultTrackers) -> String {
        var link = "magnet:?xt=urn:btih:\(infoHash)&dn=\(formEncode(name))"
        for tracker in trackers { link += "&tr=\(formEncode(tracker))" }
        return link
    }

    private static func base32ToHex(_ input: String) -> String? {
        let alphabet = Array("ABCDEFGHIJKLMNOPQRSTUVWXYZ234567")
        var buffer = 0
        var bits = 0
        var bytes: [UInt8] = []
        for char in input {
            guard let index = alphabet.firstIndex(of: char) else { return nil }
            buffer = ((buffer << 5) | index) & 0xFFFF
            bits += 5
            if bits >= 8 {
                bits -= 8
                bytes.append(UInt8((buffer >> bits) & 0xFF))
            }
        }
        guard bytes.count == 20 else { return nil }
        return bytes.map { String(format: "%02x", $0) }.joined()
    }
}
```

```swift
import Foundation

public enum ReleaseNameParser {
    public static func parse(_ raw: String) -> ParsedRelease {
        var r = ParsedRelease()
        var name = raw.replacingOccurrences(of: "_", with: ".")
        if let m = name.firstMatch(of: /^\s*\[[^\]]*\]\s*/) { name.removeSubrange(m.range) }
        if let m = name.firstMatch(of: /(?i)\.(mkv|mp4|avi|m4v|torrent)$/) { name.removeSubrange(m.range) }

        var cut = name.endIndex
        func mark(_ index: String.Index) { if index < cut { cut = index } }

        // Season / episode
        if let m = name.firstMatch(of: /(?i)\bS(\d{1,2})[ .]?E(\d{1,3})(?:-?E(\d{1,3}))?\b/) {
            r.season = Int(m.1); r.episode = Int(m.2); r.episodeEnd = m.3.flatMap { Int($0) }
            mark(m.range.lowerBound)
        } else if let m = name.firstMatch(of: /(?i)\b(\d{1,2})x(\d{2,3})\b/) {
            r.season = Int(m.1); r.episode = Int(m.2)
            mark(m.range.lowerBound)
        } else if let m = name.firstMatch(of: /(?i)\b(?:S|Season[ .]?)(\d{1,2})\b/) {
            r.season = Int(m.1)
            mark(m.range.lowerBound)
        }

        // Resolution
        if let m = name.firstMatch(of: /(?i)\b(2160|1080|720|480)[pi]\b/) {
            r.resolution = Resolution(rawValue: Int(m.1) ?? 0) ?? .unknown
            mark(m.range.lowerBound)
        } else if let m = name.firstMatch(of: /(?i)\b(?:4K|UHD)\b/) {
            r.resolution = .uhd2160
            mark(m.range.lowerBound)
        }

        // Source, codec, audio: first matching rule wins
        let sourceRules: [(Regex<Substring>, String)] = [
            (/(?i)\bREMUX\b/, "REMUX"), (/(?i)\bWEB[ .-]?DL\b/, "WEB-DL"), (/(?i)\bWEB[ .-]?Rip\b/, "WEBRip"),
            (/(?i)\b(?:Blu[ .-]?Ray|BDRip|BRRip)\b/, "BluRay"), (/(?i)\bHDTV\b/, "HDTV"),
            (/(?i)\bDVD[ .-]?Rip\b/, "DVDRip"), (/(?i)\bHDRip\b/, "HDRip"),
            (/(?i)\b(?:CAM|HDCAM|TS|TELESYNC|HDTS)\b/, "CAM"), (/(?i)\bWEB\b/, "WEB"),
        ]
        let codecRules: [(Regex<Substring>, String)] = [
            (/(?i)\b(?:x265|HEVC|H[ .]?265)\b/, "H.265"), (/(?i)\b(?:x264|AVC|H[ .]?264)\b/, "H.264"),
            (/(?i)\bAV1\b/, "AV1"), (/(?i)\bXviD\b/, "XviD"),
        ]
        let audioRules: [(Regex<Substring>, String)] = [
            (/(?i)\bAtmos\b/, "Atmos"), (/(?i)\bTrueHD\b/, "TrueHD"), (/(?i)\bDTS[ .-]?HD\b/, "DTS-HD"),
            (/(?i)\bDTS\b/, "DTS"), (/(?i)\b(?:DDP|DD\+|EAC3)/, "DD+"),
            (/(?i)\b(?:DD|AC3)(?:[ .]?\d[ .]\d)?\b/, "DD"), (/(?i)\bAAC/, "AAC"),
        ]
        func firstRule(_ rules: [(Regex<Substring>, String)]) -> String? {
            for (regex, label) in rules {
                if let m = name.firstMatch(of: regex) { mark(m.range.lowerBound); return label }
            }
            return nil
        }
        r.source = firstRule(sourceRules)
        r.codec = firstRule(codecRules)
        r.audio = firstRule(audioRules)

        // HDR flags
        if let m = name.firstMatch(of: /(?i)\bHDR10(?:\+|Plus)/) {
            r.hdr.append("HDR10+"); mark(m.range.lowerBound)
        } else if let m = name.firstMatch(of: /(?i)\bHDR(?:10)?\b/) {
            r.hdr.append("HDR"); mark(m.range.lowerBound)
        }
        if let m = name.firstMatch(of: /(?i)\b(?:DV|DoVi|Dolby[ .]?Vision)\b/) {
            r.hdr.append("DV"); mark(m.range.lowerBound)
        }

        // Flags
        if let m = name.firstMatch(of: /(?i)\bPROPER\b/) { r.isProper = true; mark(m.range.lowerBound) }
        if let m = name.firstMatch(of: /(?i)\bREPACK\b/) { r.isRepack = true; mark(m.range.lowerBound) }
        if let m = name.firstMatch(of: /(?i)\bEXTENDED\b/) { r.isExtended = true; mark(m.range.lowerBound) }
        if let m = name.firstMatch(of: /(?i)\bCOMPLETE\b/) { r.isComplete = true; mark(m.range.lowerBound) }

        // Release group: trailing "-GROUP", optionally followed by "[site]"
        if let m = name.firstMatch(of: /-([A-Za-z0-9]+)(?:\[[^\]]*\])?\s*$/) {
            let group = String(m.1)
            if !["dl", "rip"].contains(group.lowercased()) { r.group = group }
        }

        // Year: last 19xx/20xx before the cut that isn't at the very start
        let years = name.matches(of: /\b(19\d{2}|20\d{2})\b/)
            .filter { $0.range.lowerBound > name.startIndex && $0.range.lowerBound < cut }
        if let last = years.last {
            r.year = Int(last.1)
            mark(last.range.lowerBound)
        }

        // Title
        var title = String(name[name.startIndex..<cut])
        if !title.contains(" ") { title = title.replacingOccurrences(of: ".", with: " ") }
        title = title.trimmingCharacters(in: CharacterSet(charactersIn: " .-_([{").union(.whitespaces))
        r.title = title.split(separator: " ", omittingEmptySubsequences: true).joined(separator: " ")
        return r
    }
}
```

```swift
import Foundation

public protocol SourceProvider: Sendable {
    var id: String { get }
    func releases(for query: MediaQuery) async throws -> [Release]
}

public enum ReleaseRelevance {
    /// Filters free-text search hits down to what the query actually asked for.
    public static func matches(_ p: ParsedRelease, _ query: MediaQuery) -> Bool {
        switch query {
        case let .episode(_, _, season, episode):
            guard p.season == season, let first = p.episode else { return false }
            return first == episode || (p.episodeEnd.map { episode >= first && episode <= $0 } ?? false)
        case let .season(_, _, season):
            return p.season == season && p.episode == nil
        case let .movie(_, _, year):
            guard p.season == nil else { return false }
            if let year, let found = p.year { return abs(found - year) <= 1 }
            return true
        case .text:
            return true
        }
    }
}
```

```swift
public struct YTSProvider: SourceProvider {

    private let http: HTTPRequester

    public init(baseURL: URL = URL(string: "https://yts.mx")!, transport: any HTTPTransport) {
        self.baseURL = baseURL
        self.http = HTTPRequester(service: "yts", transport: transport)
    }

    public func releases(for query: MediaQuery) async throws -> [Release] {
        guard case let .movie(imdbID, _, _) = query else { return [] }
        var components = URLComponents(url: baseURL.appending(path: "api/v2/list_movies.json"), resolvingAgainstBaseURL: false)!
        components.queryItems = [URLQueryItem(name: "query_term", value: imdbID), URLQueryItem(name: "limit", value: "5")]
        let response = try await http.decode(Response.self, .get(components.url!), decoder: .snakeCase)
        return (response.data?.movies ?? []).flatMap { movie in
            (movie.torrents ?? []).compactMap { torrent -> Release? in
                guard let hash = Magnet.normalize(torrent.hash) else { return nil }
                let kind = (torrent.type ?? "web").uppercased()
                let name = "\(movie.title) (\(movie.year)) [\(torrent.quality)] [\(kind)] [YTS]"
                var parsed = ReleaseNameParser.parse(name)
                parsed.title = movie.title
                parsed.codec = torrent.videoCodec?.lowercased() == "x265" ? "H.265" : "H.264"
                return Release(name: name, link: Magnet.make(infoHash: hash, name: movie.title), infoHash: hash,
                               sizeBytes: torrent.sizeBytes, seeders: torrent.seeds,
                               sources: [SourceTag(.direct, "YTS")], parsed: parsed)
            }
        }
    }

    struct Response: Decodable {
        let data: Body?
        struct Body: Decodable { let movies: [Movie]? }
        struct Movie: Decodable { let title: String; let year: Int; let torrents: [Torrent]? }
        struct Torrent: Decodable {
            let hash: String
            let quality: String
            let type: String?
            let videoCodec: String?
            let seeds: Int?
            let sizeBytes: Int64?
        }
    }
}
```

```swift
import Foundation

public struct EZTVProvider: SourceProvider {
    public let id = "eztv"
    private let baseURL: URL
    private let http: HTTPRequester
    private let maxPages: Int
    private let pageSize = 100

    public init(baseURL: URL = URL(string: "https://eztvx.to")!, transport: any HTTPTransport, maxPages: Int = 3) {
        self.baseURL = baseURL
        self.http = HTTPRequester(service: "eztv", transport: transport)
        self.maxPages = maxPages
    }

    public func releases(for query: MediaQuery) async throws -> [Release] {
        let imdbID: String
        let season: Int
        let episode: Int?
        switch query {
        case let .episode(id, _, s, e): imdbID = id; season = s; episode = e
        case let .season(id, _, s): imdbID = id; season = s; episode = nil
        default: return []
        }
        let digits = imdbID.hasPrefix("tt") ? String(imdbID.dropFirst(2)) : imdbID

        var torrents: [Torrent] = []
        for page in 1...maxPages {
            var components = URLComponents(url: baseURL.appending(path: "api/get-torrents"), resolvingAgainstBaseURL: false)!
            components.queryItems = [
                URLQueryItem(name: "imdb_id", value: digits),
                URLQueryItem(name: "limit", value: String(pageSize)),
                URLQueryItem(name: "page", value: String(page)),
            ]
            let response = try await http.decode(Response.self, .get(components.url!), decoder: .snakeCase)
            let batch = response.torrents ?? []
            torrents += batch
            if batch.count < pageSize { break }
        }

        return torrents.compactMap { t -> Release? in
            let name = t.title ?? t.filename ?? ""
            let parsed = ReleaseNameParser.parse(name)
            let s = t.season?.value.map(Int.init) ?? parsed.season
            let e = t.episode?.value.map(Int.init) ?? parsed.episode
            if let episode {
                guard s == season, e == episode else { return nil }
            } else {
                guard s == season, e == 0 || e == nil || parsed.isSeasonPack else { return nil }
            }
            let hash = t.hash.flatMap(Magnet.normalize)
            let link: String
            if let magnet = t.magnetUrl, !magnet.isEmpty {
                link = magnet
            } else if let hash {
                link = Magnet.make(infoHash: hash, name: name)
            } else {
                return nil
            }
            return Release(name: name, link: link, infoHash: hash, sizeBytes: t.sizeBytes?.value,
                           seeders: t.seeds?.value.map(Int.init), sources: [SourceTag(.direct, "EZTV")], parsed: parsed)
        }
    }

    struct Response: Decodable { let torrents: [Torrent]? }
    struct Torrent: Decodable {
        let hash: String?
        let filename: String?
        let title: String?
        let season: LossyInt64?
        let episode: LossyInt64?
        let seeds: LossyInt64?
        let sizeBytes: LossyInt64?
        let magnetUrl: String?
    }
}
```

```swift
            guard ReleaseRelevance.matches(parsed, query) else { return nil }

        }
```

```swift
import Foundation

public struct ChillIndexer: Decodable, Sendable, Hashable {
    public let id: String
    public let name: String
    public let enabled: Bool?
}

public actor ChillProvider: SourceProvider {
    public nonisolated let id = "chill"
    private nonisolated let token: String
    private nonisolated let baseURL: URL
    private nonisolated let http: HTTPRequester
    private var cachedIndexers: [ChillIndexer]?

    public init(token: String, transport: any HTTPTransport, baseURL: URL = URL(string: "https://api.chill.institute/v4")!) {
        self.token = token
        self.baseURL = baseURL
        self.http = HTTPRequester(service: "chill", transport: transport)
    }

    public func indexers() async throws -> [ChillIndexer] {
        if let cachedIndexers { return cachedIndexers }
        let response: IndexersResponse = try await call("GetIndexers", Empty())
        let enabled = (response.indexers ?? []).filter { $0.enabled == true }
        cachedIndexers = enabled
        return enabled
    }

    public func releases(for query: MediaQuery) async throws -> [Release] {
        switch query {
        case .movie, .text:
            let text = query.searchText
            let indexers = try await indexers()
            let releases = try await withThrowingTaskGroup(of: [Release].self) { group in
                for indexer in indexers {
                    group.addTask { try await self.search(text, indexerID: indexer.id) }
                }
                var all: [Release] = []
                var lastError: (any Error)?
                while let result = await group.nextResult() {
                    switch result {
                    case let .success(rows): all += rows
                    case let .failure(error): lastError = error
                    }
                }
                if all.isEmpty, let lastError { throw lastError }
                return all
            }
            return releases.filter { ReleaseRelevance.matches($0.parsed, query) }
        case let .episode(imdbID, _, season, episode):
            let response: EpisodeResponse = try await call(
                "GetTVShowEpisodeDownload", EpisodeRequest(imdbId: imdbID, seasonNumber: season, episodeNumber: episode))
            return response.download.map { [Self.release(from: $0)] } ?? []
        case let .season(imdbID, _, season):
            let response: SeasonResponse = try await call(
                "GetTVShowSeasonDownloads", SeasonRequest(imdbId: imdbID, seasonNumber: season))
            return response.seasonPack.map { [Self.release(from: $0)] } ?? []
        }
    }

    private nonisolated func search(_ text: String, indexerID: String) async throws -> [Release] {
        let response: SearchResponse = try await call("Search", SearchRequest(query: text, indexerId: indexerID))
        return (response.results ?? []).compactMap(Self.release(from:))
    }

    private nonisolated func call<Req: Encodable & Sendable, Res: Decodable>(_ method: String, _ body: Req) async throws -> Res {
        let request = try URLRequest.postJSON(
            baseURL.appending(path: "chill.v4.UserService/\(method)"), body: body,
            headers: ["Authorization": "Bearer \(token)", "Connect-Protocol-Version": "1"])
        return try await http.decode(Res.self, request)
    }

    static func release(from result: SearchResult) -> Release? {
        guard let link = result.link, !link.isEmpty else { return nil }
        let name = result.title ?? ""
        var parsed = ReleaseNameParser.parse(name)
        if let info = result.releaseInfo {
            if let title = info.title, !title.isEmpty { parsed.title = title }
            if let year = info.year { parsed.year = year }
            if let season = info.season { parsed.season = season }
            if let episode = info.episode { parsed.episode = episode }
            if let label = info.resolution, Resolution(label: label) != .unknown { parsed.resolution = Resolution(label: label) }
        }
        let tagName = [result.source, result.indexer].compactMap { $0 }.first { !$0.isEmpty } ?? "chill"
        return Release(name: name, link: link, sizeBytes: result.size?.value,
                       seeders: result.seeders?.value.map(Int.init), sources: [SourceTag(.chill, tagName)], parsed: parsed)
    }

    static func release(from download: TVShowDownload) -> Release {
        var parsed = ReleaseNameParser.parse(download.title ?? "")
        if let label = download.resolution, Resolution(label: label) != .unknown { parsed.resolution = Resolution(label: label) }
        return Release(name: download.title ?? "", link: download.link ?? "", sizeBytes: download.size?.value,
                       seeders: download.seeders?.value.map(Int.init),
                       sources: [SourceTag(.chill, download.indexer ?? "chill")], parsed: parsed)
    }

    struct Empty: Encodable, Sendable {}
    struct SearchRequest: Encodable, Sendable { let query: String; let indexerId: String }
    struct EpisodeRequest: Encodable, Sendable { let imdbId: String; let seasonNumber: Int; let episodeNumber: Int }
    struct SeasonRequest: Encodable, Sendable { let imdbId: String; let seasonNumber: Int }

    struct IndexersResponse: Decodable { let indexers: [ChillIndexer]? }
    struct SearchResponse: Decodable { let results: [SearchResult]? }
    struct SearchResult: Decodable {
        let title: String?
        let indexer: String?
        let link: String?
        let seeders: LossyInt64?
        let size: LossyInt64?
        let source: String?
        let releaseInfo: ReleaseInfo?
    }
    struct ReleaseInfo: Decodable {
        let title: String?
        let year: Int?
        let season: Int?
        let episode: Int?
        let resolution: String?
    }
    struct TVShowDownload: Decodable {
        let title: String?
        let link: String?
        let size: LossyInt64?
        let seeders: LossyInt64?
        let resolution: String?
        let indexer: String?
    }
    struct EpisodeResponse: Decodable { let download: TVShowDownload? }
    struct SeasonResponse: Decodable { let seasonPack: TVShowDownload? }
}
```

```swift
import Foundation

public struct ProviderFailure: Sendable, Equatable {
    public let providerID: String
    public let error: AppError
    public init(providerID: String, error: AppError) { self.providerID = providerID; self.error = error }
}

public struct ReleaseSnapshot: Sendable, Equatable {
    public var releases: [Release] = []
    public var failures: [ProviderFailure] = []
    public var pendingProviders: Set<String> = []
    public var isComplete: Bool { pendingProviders.isEmpty }
    public init() {}
}

public struct ReleaseAggregator: Sendable {
    private let providers: [any SourceProvider]
    private let timeout: Duration

    public init(providers: [any SourceProvider], timeout: Duration = .seconds(12)) {
        self.providers = providers
        self.timeout = timeout
    }

    public func stream(for query: MediaQuery) -> AsyncStream<ReleaseSnapshot> {
        let providers = providers
        let timeout = timeout
        return AsyncStream { continuation in
            let task = Task {
                var snapshot = ReleaseSnapshot()
                snapshot.pendingProviders = Set(providers.map(\.id))
                continuation.yield(snapshot)
                await withTaskGroup(of: (String, Result<[Release], AppError>).self) { group in
                    for provider in providers {
                        group.addTask { (provider.id, await Self.run(provider, query, timeout)) }
                    }
                    for await (id, result) in group {
                        snapshot.pendingProviders.remove(id)
                        switch result {
                        case let .success(rows): snapshot.releases = Self.merge(snapshot.releases, rows)
                        case let .failure(error): snapshot.failures.append(ProviderFailure(providerID: id, error: error))
                        }
                        continuation.yield(snapshot)
                    }
                }
                continuation.finish()
            }
            continuation.onTermination = { _ in task.cancel() }
        }
    }

    public func collect(for query: MediaQuery) async -> ReleaseSnapshot {
        var last = ReleaseSnapshot()
        for await snapshot in stream(for: query) { last = snapshot }
        return last
    }

    static func run(_ provider: any SourceProvider, _ query: MediaQuery, _ timeout: Duration) async -> Result<[Release], AppError> {
        do {
            let rows = try await withThrowingTaskGroup(of: [Release].self) { group in
                group.addTask { try await provider.releases(for: query) }
                group.addTask {
                    try await Task.sleep(for: timeout)
                    throw AppError.timeout(service: provider.id)
                }
                defer { group.cancelAll() }
                return try await group.next() ?? []
            }
            return .success(rows)
        } catch let error as AppError {
            return .failure(error)
        } catch {
            return .failure(.transport("\(provider.id): \(error.localizedDescription)"))
        }
    }

    public static func merge(_ existing: [Release], _ incoming: [Release]) -> [Release] {
        var order: [String] = []
        var byID: [String: Release] = [:]
        for row in existing + incoming {
            guard var current = byID[row.id] else {
                order.append(row.id)
                byID[row.id] = row
                continue
            }
            current.sources.formUnion(row.sources)
            if let seeders = row.seeders { current.seeders = max(current.seeders ?? 0, seeders) }
            if current.sizeBytes == nil { current.sizeBytes = row.sizeBytes }
            if !current.link.hasPrefix("magnet:"), row.link.hasPrefix("magnet:") { current.link = row.link }
            byID[row.id] = current
        }
        return order.compactMap { byID[$0] }
    }
}

public enum ReleaseSort: String, Sendable, CaseIterable { case seeders, size, quality }

extension Array where Element == Release {
    public func sorted(by sort: ReleaseSort) -> [Release] {
        switch sort {
        case .seeders:
            sorted { ($0.seeders ?? -1, $0.parsed.resolution) > ($1.seeders ?? -1, $1.parsed.resolution) }
        case .size:
            sorted { ($0.sizeBytes ?? -1) > ($1.sizeBytes ?? -1) }
        case .quality:
            sorted { ($0.parsed.resolution, $0.seeders ?? -1) > ($1.parsed.resolution, $1.seeders ?? -1) }
        }
    }

    public func filtered(resolution: Resolution?) -> [Release] {
        guard let resolution else { return self }
        return filter { $0.parsed.resolution == resolution }
    }
}
```

```swift
import Foundation

public struct PutioClient: Sendable {
    public static let defaultBaseURL = URL(string: "https://api.put.io/v2")!
    private let token: String
    private let baseURL: URL
    private let http: HTTPRequester

    public init(token: String, transport: any HTTPTransport, baseURL: URL = PutioClient.defaultBaseURL) {
        self.token = token
        self.baseURL = baseURL
        self.http = HTTPRequester(service: "putio", transport: transport)
    }

    // MARK: Auth

    public static func authorizeURL(clientID: String, redirectURI: String) -> URL {
        URL(string: "\(defaultBaseURL.absoluteString)/oauth2/authenticate?client_id=\(clientID)&response_type=token&redirect_uri=\(redirectURI)")!
    }

    public static func token(fromCallback url: URL) -> String? {
        guard let fragment = url.fragment else { return nil }
        var components = URLComponents()
        components.query = fragment
        return components.queryItems?.first { $0.name == "access_token" }?.value.flatMap { $0.isEmpty ? nil : $0 }
    }

    // MARK: Account

    public func accountInfo() async throws -> PutioAccount {
        try await get("account/info", as: InfoEnvelope.self).info
    }

    // MARK: Transfers

    public func addTransfer(url: String, saveParentID: Int64?) async throws -> PutioTransfer {
        var fields = [("url", url)]
        if let saveParentID { fields.append(("save_parent_id", String(saveParentID))) }
        return try await post("transfers/add", fields, as: TransferEnvelope.self).transfer
    }

    public func transfer(id: Int64) async throws -> PutioTransfer {
        try await get("transfers/\(id)", as: TransferEnvelope.self).transfer
    }

    public func transfers() async throws -> [PutioTransfer] {
        try await get("transfers/list", as: TransfersEnvelope.self).transfers
    }

    public func cancelTransfers(ids: [Int64]) async throws {
        _ = try await http.data(form("transfers/cancel", [("transfer_ids", ids.map(String.init).joined(separator: ","))]))
    }

    public func retryTransfer(id: Int64) async throws -> PutioTransfer {
        try await post("transfers/retry", [("id", String(id))], as: TransferEnvelope.self).transfer
    }

    // MARK: Files

    public func listFiles(parentID: Int64) async throws -> PutioFileList {
        try await get("files/list", query: [("parent_id", String(parentID)), ("per_page", "1000")], as: PutioFileList.self)
    }

    public func file(id: Int64) async throws -> PutioFile {
        try await get("files/\(id)", as: FileEnvelope.self).file
    }

    public func createFolder(name: String, parentID: Int64) async throws -> PutioFile {
        try await post("files/create-folder", [("name", name), ("parent_id", String(parentID))], as: FileEnvelope.self).file
    }

    public func startFrom(fileID: Int64) async throws -> Int {
        try await get("files/\(fileID)/start-from", as: StartFromEnvelope.self).startFrom ?? 0
    }

    public func setStartFrom(fileID: Int64, seconds: Int) async throws {
        _ = try await http.data(form("files/\(fileID)/start-from/set", [("time", String(seconds))]))
    }

    public func originalURL(fileID: Int64) async throws -> URL {
        let envelope = try await get("files/\(fileID)/url", as: URLEnvelope.self)
        guard let url = URL(string: envelope.url) else { throw AppError.decoding(service: "putio", detail: "bad url") }
        return url
    }

    public func subtitles(fileID: Int64) async throws -> [PutioSubtitle] {
        try await get("files/\(fileID)/subtitles", as: SubtitlesEnvelope.self).subtitles
    }

    public func hlsURL(fileID: Int64) -> URL {
        URL(string: "\(baseURL.absoluteString)/files/\(fileID)/hls/media.m3u8?subtitle_key=all&oauth_token=\(formEncode(token))")!
    }

    // MARK: Plumbing

    private var authHeaders: [String: String] { ["Authorization": "token \(token)"] }

    private func get<T: Decodable>(_ path: String, query: [(String, String)] = [], as: T.Type) async throws -> T {
        var components = URLComponents(url: baseURL.appending(path: path), resolvingAgainstBaseURL: false)!
        if !query.isEmpty { components.queryItems = query.map { URLQueryItem(name: $0.0, value: $0.1) } }
        return try await http.decode(T.self, .get(components.url!, headers: authHeaders), decoder: .snakeCase)
    }

    private func form(_ path: String, _ fields: [(String, String)]) -> URLRequest {
        .postForm(baseURL.appending(path: path), fields: fields, headers: authHeaders)
    }

    private func post<T: Decodable>(_ path: String, _ fields: [(String, String)], as: T.Type) async throws -> T {
        try await http.decode(T.self, form(path, fields), decoder: .snakeCase)
    }

    struct InfoEnvelope: Decodable { let info: PutioAccount }
    struct TransferEnvelope: Decodable { let transfer: PutioTransfer }
    struct TransfersEnvelope: Decodable { let transfers: [PutioTransfer] }
    struct FileEnvelope: Decodable { let file: PutioFile }
    struct StartFromEnvelope: Decodable { let startFrom: Int? }
    struct URLEnvelope: Decodable { let url: String }
    struct SubtitlesEnvelope: Decodable { let subtitles: [PutioSubtitle] }
}
```

```swift
import Foundation

public enum MediaKind: String, Sendable, Codable, Hashable { case movie, tv, person }

public struct CatalogItem: Sendable, Hashable, Identifiable, Decodable {
    public let tmdbID: Int
    public let mediaType: MediaKind
    public let title: String
    public let overview: String
    public let posterPath: String?
    public let backdropPath: String?
    public let profilePath: String?
    public let year: Int?
    public let voteAverage: Double?
    public let genreIDs: [Int]
    public let knownFor: [CatalogItem]

    public var id: String { "\(mediaType.rawValue)-\(tmdbID)" }

    // Keys are camelCase because decoding uses .convertFromSnakeCase.
    enum CodingKeys: String, CodingKey {
        case id, mediaType, title, name, overview, posterPath, backdropPath, profilePath
        case releaseDate, firstAirDate, voteAverage, genreIds, knownFor
    }

    public init(from decoder: any Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        tmdbID = try c.decode(Int.self, forKey: .id)
        let movieTitle = try c.decodeIfPresent(String.self, forKey: .title)
        let name = try c.decodeIfPresent(String.self, forKey: .name)
        mediaType = try c.decodeIfPresent(MediaKind.self, forKey: .mediaType) ?? (movieTitle != nil ? .movie : .tv)
        title = movieTitle ?? name ?? ""
        overview = try c.decodeIfPresent(String.self, forKey: .overview) ?? ""
        posterPath = try c.decodeIfPresent(String.self, forKey: .posterPath)
        backdropPath = try c.decodeIfPresent(String.self, forKey: .backdropPath)
        profilePath = try c.decodeIfPresent(String.self, forKey: .profilePath)
        let date = try c.decodeIfPresent(String.self, forKey: .releaseDate) ?? c.decodeIfPresent(String.self, forKey: .firstAirDate)
        year = date.flatMap { Int($0.prefix(4)) }
        voteAverage = try c.decodeIfPresent(Double.self, forKey: .voteAverage)
        genreIDs = try c.decodeIfPresent([Int].self, forKey: .genreIds) ?? []
        knownFor = try c.decodeIfPresent([CatalogItem].self, forKey: .knownFor) ?? []
    }
}
```

```swift
import Foundation

public protocol CatalogSearching: Sendable {
    func searchMovie(title: String, year: Int?) async throws -> [CatalogItem]
    func searchTV(title: String, year: Int?) async throws -> [CatalogItem]
}

extension CatalogClient: CatalogSearching {}

public struct LibraryMatch: Sendable, Equatable, Hashable {
    public let tmdbID: Int
    public let mediaType: MediaKind
    public let season: Int?
    public let episode: Int?
    public let confidence: Double

    public init(tmdbID: Int, mediaType: MediaKind, season: Int?, episode: Int?, confidence: Double) {
        self.tmdbID = tmdbID; self.mediaType = mediaType; self.season = season; self.episode = episode; self.confidence = confidence
    }
}

public struct LibraryMatcher: Sendable {
    private let catalog: any CatalogSearching

    public init(catalog: any CatalogSearching) { self.catalog = catalog }

    public func match(fileName: String) async throws -> LibraryMatch? {
        let parsed = ReleaseNameParser.parse(fileName)
        guard !parsed.title.isEmpty else { return nil }
        let isTV = parsed.season != nil
        let results = isTV
            ? try await catalog.searchTV(title: parsed.title, year: nil)
            : try await catalog.searchMovie(title: parsed.title, year: parsed.year)
        guard let best = Self.pick(results, title: parsed.title, year: isTV ? nil : parsed.year) else { return nil }
        return LibraryMatch(tmdbID: best.item.tmdbID, mediaType: isTV ? .tv : .movie,
                            season: parsed.season, episode: parsed.episode, confidence: best.confidence)
    }

    public static func normalize(_ text: String) -> String {
        text.folding(options: [.diacriticInsensitive, .caseInsensitive], locale: .init(identifier: "en_US"))
            .replacingOccurrences(of: "&", with: "and")
            .filter { $0.isLetter || $0.isNumber }
            .lowercased()
    }

    public static func pick(_ items: [CatalogItem], title: String, year: Int?) -> (item: CatalogItem, confidence: Double)? {
        let wanted = normalize(title)
        func yearOK(_ item: CatalogItem) -> Bool {
            guard let year else { return true }
            guard let found = item.year else { return false }
            return abs(found - year) <= 1
        }
        if let exact = items.first(where: { normalize($0.title) == wanted && yearOK($0) }) { return (exact, 1.0) }
        if let titleOnly = items.first(where: { normalize($0.title) == wanted }) { return (titleOnly, 0.7) }
        if year != nil, let first = items.first, yearOK(first) { return (first, 0.5) }
        return nil
    }
}
```

```swift
import Foundation

public enum TransferPhase: Sendable, Equatable {
    case queued
    case downloading
    case finishing
    case completed(fileID: Int64?)
    case failed(message: String)

    public init(_ transfer: PutioTransfer) {
        switch (transfer.status ?? "").uppercased() {
        case "COMPLETED", "SEEDING": self = .completed(fileID: transfer.fileId)
        case "ERROR": self = .failed(message: transfer.errorMessage ?? "put.io reported an error")
        case "DOWNLOADING", "PREPARING_DOWNLOAD": self = .downloading
        case "COMPLETING": self = .finishing
        default: self = .queued
        }
    }
}

/// Flags a transfer that is downloading with zero peers and no progress for `threshold` seconds.
public struct StallDetector: Sendable {
    public let threshold: TimeInterval
    private var lastProgress: Int64?
    private var lastProgressAt: Date?

    public init(threshold: TimeInterval = 180) { self.threshold = threshold }

    public mutating func isStalled(_ transfer: PutioTransfer, now: Date) -> Bool {
        guard TransferPhase(transfer) == .downloading else {
            lastProgress = nil
            lastProgressAt = nil
            return false
        }
        let progress = transfer.downloaded ?? Int64(transfer.percentDone ?? 0)
        guard progress == lastProgress, let since = lastProgressAt else {
            lastProgress = progress
            lastProgressAt = now
            return false
        }
        return (transfer.peersConnected ?? 0) == 0 && now.timeIntervalSince(since) >= threshold
    }
}
```

```swift
import Foundation

/// Debounced, cancel-on-newer search with per-query caching and a one-word-shorter retry.
public actor LiveSearchEngine<Item: Sendable> {
    private let debounce: Duration
    private let search: @Sendable (String) async throws -> [Item]
    private var generation = 0
    private var cache: [String: [Item]] = [:]

    public init(debounce: Duration = .milliseconds(250), search: @escaping @Sendable (String) async throws -> [Item]) {
        self.debounce = debounce
        self.search = search
    }

    /// Returns nil when a newer call superseded this one.
    public func results(for raw: String) async throws -> [Item]? {
        generation += 1
        let mine = generation
        let query = raw.trimmingCharacters(in: .whitespacesAndNewlines)
        let key = query.lowercased()
        guard !query.isEmpty else { return [] }
        if let hit = cache[key] { return hit }

        if debounce > .zero { try await Task.sleep(for: debounce) }
        guard mine == generation else { return nil }

        var items = try await search(query)
        guard mine == generation else { return nil }
        if items.isEmpty, let shorter = Self.dropLastWord(query) {
            items = try await search(shorter)
            guard mine == generation else { return nil }
        }
        cache[key] = items
        return items
    }

    static func dropLastWord(_ text: String) -> String? {
        let words = text.split(separator: " ")
        guard words.count > 1 else { return nil }
        return words.dropLast().joined(separator: " ")
    }
}
```

```swift
    var visited = 0
```
